function getGradeStyle(gradeNum) {
    if (gradeNum === 1 || gradeNum === 2) return "highlight-green";

function runAnalysis() {
    let gender = document.getElementById('gender').value;

    let data = {
        bf: {name: "체지방률", v1: parseFloat(document.getElementById('bf_1').value), v2: parseFloat(document.getElementById('bf_2').value), unit: "%", inv: true},

        sf: {name: "어깨 뒤로 손잡기", v1: parseFloat(document.getElementById('sf_1').value), v2: parseFloat(document.getElementById('sf_2').value), unit: "cm", inv: true}
    };

    function calcGrade(key, val) {
        if (isNaN(val) || val === 0) return 5;
        if (key === 'bf') {
            return val <= 18.4 ? 1 : (val <= 22.3 ? 2 : (val <= 26.1 ? 3 : (val <= 29.9 ? 4 : 5)));
        } else if (key === 'sh') {
            return val >= 69 ? 1 : (val >= 50 ? 2 : (val >= 40 ? 3 : (val >= 30 ? 4 : 5)));
        } else if (key === 'bn') {
            return val >= 78 ? 1 : (val >= 66 ? 2 : (val >= 54 ? 3 : (val >= 42 ? 4 : 5)));
        } else if (key === 'sq' || key === 'de') {
            return val >= 134 ? 1 : (val >= 94 ? 2 : (val >= 74 ? 3 : (val >= 54 ? 4 : 5)));
        } else if (key === 'lp') {
            return val >= 163 ? 1 : (val >= 136 ? 2 : (val >= 109 ? 3 : (val >= 82 ? 4 : 5)));
        } else if (key === 'pu1') {
            return val >= 40 ? 1 : (val >= 30 ? 2 : (val >= 20 ? 3 : (val >= 10 ? 4 : 5)));
        } else if (key === 'pu2') {
            return val >= 55 ? 1 : (val >= 43 ? 2 : (val >= 31 ? 3 : (val >= 19 ? 4 : 5)));
        } else if (key === 'hg') {
            return val >= 60 ? 1 : (val >= 45 ? 2 : (val >= 30 ? 3 : (val >= 15 ? 4 : 5)));
        } else if (key === 'fl') {
            return val >= 24 ? 1 : (val >= 19 ? 2 : (val >= 14 ? 3 : (val >= 2 ? 4 : 5)));
        } else if (key === 'sf') {
            return val <= -5 ? 1 : (val <= 0 ? 2 : (val <= 5 ? 3 : (val <= 10 ? 4 : 5)));
        }
        return 3;
    }

    let summaryHtml = `<tr>

        <th>시각적 변화 (그래프)</th>
    </tr>`;

    let keys = ['bf', 'sh', 'bn', 'sq', 'de', 'lp', 'hg', 'fl', 'sf'];
    keys.forEach(k => {
        let item = data[k];
        let g1 = calcGrade(k, item.v1);
        let g2 = calcGrade(k, item.v2);
        
        let diff = item.v2 - item.v1;
        let rate = item.v1 !== 0 ? ((diff / item.v1) * 100).toFixed(1) : 0;
        let diffStr = (diff > 0 ? "+" + diff.toFixed(1) : diff.toFixed(1)) + item.unit;
        let rateStr = (item.inv ? (diff <= 0 ? "감소 (개선)" : "증가 (악화)") : (diff >= 0 ? "+" + rate + "%" : rate + "%"));
        let valClass = (item.inv ? diff <= 0 : diff >= 0) ? "plus-val" : "minus-val";

        let maxVal = Math.max(Math.abs(item.v1), Math.abs(item.v2)) * 1.3 || 100;
